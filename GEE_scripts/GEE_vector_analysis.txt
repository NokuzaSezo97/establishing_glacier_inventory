// Now, let's learn one technique which is commonly used, value extraction using
// vector features.

// In, GLIMS, we don't have information about the slope and aspect information to learn.
// So we're using SRTM elevation data to derive degree slope and aspect layers,
// and extract the zonal values.

// First, let's call the SRTM data from the data catalog

// Then, the next step is to apply API functions of calculation of degree slope

var slp = ee.Terrain.slope(elevation).clip(aoi).rename('SLOPE'); // Notice that I apply image clipping API function

print('Slope', slp);

// We can visualize slope layer in Map.

Map.addLayer(slp, {}, 'Slope');

// Applying zonal extraction

// Note that I'm still using the filtered GLIMS product with focused polygons at
// highly elevated zone.

glims = slp.select('SLOPE').reduceRegions({
  collection: glims,
  reducer: ee.Reducer.mean(),
  scale: 30
}).map(function(feature){
  return feature.set('SLOPE', feature.get('mean'));
});


print('Extracted GLIMS2', glims.limit(10));

// This time, we will visualize slope distribution

var chart4 = ui.Chart.feature.histogram({
  features: glims,
  property: 'SLOPE'
})
.setOptions({
  title: 'Slope Distribution',
  hAxis: {
    title: 'Slope (Degrees)'
  },
  vAxis: {
    title: 'Number of Glaciers'
  }
});

print('Slope Histogram', chart4);
