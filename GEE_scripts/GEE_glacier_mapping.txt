// FUNCTION FOR QUALITY MASKING FOR LANDSAT IMAGERY (BOTH L5 AND L8)
///////////////////////////////////////////////////////////////////////

function QAMASK(img){
  // Preprocessing all the ee.Images available in country boundary
  // 
  var qa = img.select('QA_PIXEL');
  
  // selecting the mask bits
  var dil_cloud = qa.rightShift(1).bitwiseAnd(1);
  var cirrus = qa.rightShift(2).bitwiseAnd(1);
  var cloud = qa.rightShift(3).bitwiseAnd(1);
  var cloudshadow = qa.rightShift(4).bitwiseAnd(1);
  
  // keeping the valid mask
  var mask = dil_cloud.eq(0).and(cirrus.eq(0)).and(cloud.eq(0)).and(cloudshadow.eq(0));
  
  var output = img.updateMask(mask).copyProperties(img, ['system:time_start', 'WRS_PATHROW']);
  
  return ee.Image(output);
}

//// Note: This function is exclusively for Landsat-8 image collection.
//// Uncomment this code section if you're using Landsat-8 for 2025

function PREPROCESS(col, operation){
  
  var revised = col.map(function(img){
    var img2 = img.select('SR_B.*');
    
    img2 = (img2.multiply(0.0000275)).subtract(0.2);
    
    // img2 = img2.focalMean(1, 'square','pixels', 3);

    var ndsi = img2.normalizedDifference(['SR_B3', 'SR_B6']).rename('NDSI');
    img2 = img2.addBands(ndsi);

    // Conversion for thermal Band
    var img3 = img.select('ST_B10');
    img3 = (img3.multiply(0.00341802)).add(149).toFloat().rename('ST_B10');
    img2 = img2.addBands(img3);
    
    return ee.Image(img2.copyProperties(img, ['system:time_start', 'WRS_PATHROW']));
  });
  
  var median;
  if (operation == 'min'){
    median = revised.min().clip(aoi);
  }else if (operation == 'max'){
    median = revised.max().clip(aoi);
  }else{
    median = revised.mean().clip(aoi);
  }
  
  
  return ee.Image(median.copyProperties(revised, ['system:time_start', 'WRS_PATHROW']));
}

  variablesPerSplit: 2,
  minLeafPopulation: 1
}).train({
  features: train_set, // PAY ATTENTION TO THIS
  classProperty: 'LC',
  inputProperties: img_comp.bandNames()
});
